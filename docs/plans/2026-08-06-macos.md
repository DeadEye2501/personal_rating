# Приложение под macOS — план реализации

> **Исполнителю:** слайс — одна задача. После каждой задачи ревью отдельными
> агентами по измерению, чистка, коммит; следующая начинается без паузы.
> Порядок работы — `CLAUDE.md`, раздел «Порядок работы». Шаги отмечены
> `- [ ]` для отслеживания.

**Цель:** то же приложение работает на macOS и раздаётся двумя `.dmg`, а
Windows-версия не меняется ни в поведении, ни в сборке.

**Как устроено:** различий между системами ровно три — где живут данные, чем
рисуется окно, какая иконка. Все три отвечает один новый модуль `bridge/system.py`,
и **платформа приходит в него параметром**, поэтому обе ветки сторожатся с
Windows. Командная строка PyInstaller заменяется `.spec`-файлом: там список
ресурсов — питоновский кортеж, и виндовый разделитель `;` исчезает сам.

**Стек:** Python 3.13, pywebview, PyInstaller, pytest. Спека —
`docs/specs/2026-08-06-macos.md`, замеры — `docs/other/2026-08-06-macos-research.md`.

## Общие ограничения

- **Windows-поведение не меняется ни в чём.** Ветка `win32` обязана вернуть ровно
  то, что возвращается сегодня. Собранный exe проверяется запуском, а не рассуждением.
- **Фронт не трогается — ни строки.** `client/` в этом плане не появляется вовсе.
- **Бэкенд не трогается — ни строки.** `backend/` меняется только в задаче 3, и
  только в тестах.
- **Платформа приходит параметром.** Ни одна функция `bridge/system.py` не читает
  `sys.platform` изнутри; читает его точка вызова.
- Манифест чистого кода вменяется всегда. Тест пишется до кода и хоть раз виден
  красным.
- Папка данных на macOS: `~/Library/Application Support/Personal Rating`.
- Движки: `win32` → `edgechromium`, `darwin` → `wkwebview`.
- Нижняя планка macOS — `11.0`. Идентификатор бандла — `ru.deadeye.personalrating`.
- Прогон питона: `pipenv run pytest -q`. Прогон фронта в этом плане не нужен.

"""Чем эта операционная система отличается от других.

Платформа приходит параметром, а не читается изнутри: иначе маковская ветка не
наблюдается с Windows ничем и разъезжается молча. Читает `sys.platform` точка
вызова — `main.py` и `bridge/window.py`."""

from pathlib import Path

TITLE = "Personal Rating"

# Штатное место данных приложения на macOS.
SUPPORT = ("Library", "Application Support")


def data_home(system: str, beside: Path, user: Path) -> Path:
    """Папка под `.env`, обе базы и лог.

    На macOS — своё место: писать внутрь `.app` нельзя ни технически, ни по
    правилам системы. Везде остальном — рядом с программой, как решено
    03.08.2026. Под заморозкой и из исходников на macOS папка одна и та же:
    разделять эти два случая незачем, а лишняя ветка стоила бы своего сторожа."""
    if system == "darwin":
        return user.joinpath(*SUPPORT, TITLE)
    return beside
```

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class Engine:
    name: str
    advice: str


# Список разрешённого, а не запрещённого: седьмой движок обязан упереться в
# проверку. Меняется от системы только сам список.
ENGINES = {
    "win32": Engine(
        "edgechromium",
        "среда WebView2 не установлена, и оформление работать не будет"),
    "darwin": Engine(
        "wkwebview",
        "pywebview не нашёл WKWebView, и оформление работать не будет"),
}


def engine(system: str) -> Engine | None:
    """Движок, которым эта система обязана рисовать окно.

    `None` означает «система неизвестна, требовать с неё нечего»: ронять запуск
    из-за неподдержанной системы значит менять поведение там, где сегодня
    приложение всего лишь жалуется."""
    return ENGINES.get(system)


def icon_suffix(system: str) -> str:
    """PyInstaller на macOS `.ico` отвергает на этапе сборки, а окно с ней хоть
    и поднимается, но иконка в доке выходит мыльной."""
    return ".icns" if system == "darwin" else ".ico"
```

# Тип записи → сторона картинки. Взяты только PNG-типы: старые форматы с
# отдельной маской прозрачности здесь не нужны.
TYPES = {b"ic07": 128, b"ic08": 256, b"ic09": 512, b"ic13": 256, b"ic14": 512}


def png(image: Image.Image, side: int) -> bytes:
    buffer = BytesIO()
    image.resize((side, side), Image.LANCZOS).save(buffer, format="PNG")
    return buffer.getvalue()


def main() -> None:
    source = Image.open(SRC).convert("RGBA")
    chunks = []
    for kind, side in TYPES.items():
        body = png(source, side)
        chunks.append(kind + struct.pack(">I", len(body) + 8) + body)

    payload = b"".join(chunks)
    DST.write_bytes(b"icns" + struct.pack(">I", len(payload) + 8) + payload)
    print(f"{DST}: {DST.stat().st_size} байт, записей {len(chunks)}")


if __name__ == "__main__":
    main()
```

# Сборка на две системы. Имена `Analysis`, `PYZ`, `EXE`, `COLLECT`, `BUNDLE`
# PyInstaller подставляет сам — импортировать их в spec-файле не нужно.
#
# Списком, а не командной строкой, это стало ради macOS: у `--add-data` там
# другой разделитель, и одна и та же команда не годится обеим системам.
import sys

NAME = "Personal Rating"
MACOS = sys.platform == "darwin"
ICON = f"{NAME}.icns" if MACOS else f"{NAME}.ico"

# Что зачем: `client/dist` — страница, без неё окно пустое; `migrations` с
# `alembic.ini` — схема базы; `.env.example` — из него заводится файл настроек.
DATA = [
    ("client/dist", "client/dist"),
    ("migrations", "migrations"),
    ("alembic.ini", "."),
    (".env.example", "."),
    (ICON, "."),
]

# `aiosqlite` тянется строкой подключения, `logging.config` — только из
# `migrations/env.py`, который alembic грузит на лету. Оба PyInstaller сам не
# находит.
HIDDEN = ["aiosqlite", "logging.config"]

a = Analysis(["main.py"], datas=DATA, hiddenimports=HIDDEN)
pyz = PYZ(a.pure)

if MACOS:
    # Внутри бандла `--onefile` распаковывал бы себя при каждом запуске.
    exe = EXE(pyz, a.scripts, exclude_binaries=True, name=NAME,
              console=False, icon=ICON)
    coll = COLLECT(exe, a.binaries, a.datas, name=NAME)
    app = BUNDLE(
        coll,
        name=f"{NAME}.app",
        icon=ICON,
        bundle_identifier="ru.deadeye.personalrating",
        info_plist={
            "CFBundleName": NAME,
            "CFBundleDisplayName": NAME,
            "CFBundleShortVersionString": "1.0.0",
            # Без этого страница в окне выходит мыльной на всяком современном маке.
            "NSHighResolutionCapable": True,
            "LSMinimumSystemVersion": "11.0",
        },
    )
else:
    exe = EXE(pyz, a.scripts, a.binaries, a.datas, name=NAME,
              console=False, icon=ICON, onefile=True)
```
