# План 1. Фундамент и хранилище

> **Исполнителю:** ОБЯЗАТЕЛЬНЫЙ СУБ-СКИЛЛ — `superpowers:subagent-driven-development`
> или `superpowers:executing-plans`. Шаги помечены чекбоксами (`- [ ]`) для отметки
> хода работ.

**Цель:** собрать нижний слой приложения — конфигурацию, типы границы, схему базы,
миграции со справочником жанров, репозитории и хранилище постеров, — так, чтобы всё
это проверялось `pytest` без сети, без Qt и без живых ключей.

**Архитектура:** SQLAlchemy Core в асинхронном режиме поверх `aiosqlite`, схема
объявлена императивно (`MetaData` + `Table`), а не декларативно: наружу отдаются
собственные типы из `backend/model/`, и ORM-объектам незачем существовать. Миграции
Alembic ходят синхронным драйвером, приложение — асинхронным; URL отличается только
префиксом. Постеры лежат во втором файле SQLite со своим движком.

**Стек:** Python 3.13, SQLAlchemy 2.x `[asyncio]`, aiosqlite, alembic, pytest,
pytest-asyncio.

## Глобальные ограничения

Действуют на каждую задачу, повторять в них не нужно.

- **`backend/` не импортирует `client`. `client/` не импортирует `sqlalchemy`,
  `aiosqlite`, `httpx`, `alembic`.** Держится тестом из задачи 1.
- **Через границу слоёв ходят только типы из `backend/model/`, байты, числа и
  строки.** ORM-объекты, открытые сессии и `Row` наружу не отдаются никогда.
- **Ни одного обращения к сети в тестах.** В сеть ходят только два шага
  исполнения, и оба не тесты: сбор игровых жанров (задача 4) и сверка
  идентификаторов TMDB (задача 8, шаг 1).
- **База в тестах — временный файл, свой на каждый прогон.** Не `:memory:`:
  проверяются два файла базы сразу и внешние ключи между ними, которых нет.
- **Молча не глотаем ничего.** Пойманное исключение ведёт либо к показу, либо к
  записи в лог — как минимум одно из двух.
- **Комментарий объясняет неочевидное решение и пишется коротко.** Ссылок на
  источники заимствований и декоративных рамок-разделителей в коде нет.
- **Со стороны каталога удаление каскадное**, запрещающих внешних ключей в схеме нет.
- Имена таблиц и полей — ровно те, что в `docs/specs/2026-07-27-data-model.md`.

```python
import ast
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]

BANNED_IN_CLIENT = {"sqlalchemy", "aiosqlite", "httpx", "alembic", "PIL"}

# Из бэкенда клиенту открыты только типы границы и настройки. Всё прочее —
# репозитории, сервисы, клиенты источников — тянет за собой и сессию
# SQLAlchemy, и поход к диску в главном потоке, то есть ровно то, от чего
# граница слоёв и заведена. Список корневых имён этого не ловит: они
# `backend.*`, а не `sqlalchemy`.
# Разрешение заводится под существующий модуль, а не под будущий: `backend.app`
# появится в плане 4 и будет добавлен сюда там же.
ALLOWED_FROM_BACKEND = {"backend.model", "backend.config"}


def imported_roots(path: Path):
    """Корневые имена всех модулей, которые импортирует файл."""
    tree = ast.parse(path.read_text(encoding="utf-8"))
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                yield alias.name.split(".")[0]
        # level > 0 — относительный импорт, он внутри своего же пакета
        elif isinstance(node, ast.ImportFrom) and node.level == 0 and node.module:
            yield node.module.split(".")[0]


def imported_modules(path: Path):
    """Полные имена модулей, а не только корни."""
    tree = ast.parse(path.read_text(encoding="utf-8"))
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                yield alias.name
        elif isinstance(node, ast.ImportFrom) and node.level == 0 and node.module:
            yield node.module


def offenders(package: str, banned: set[str]) -> list[str]:
    # Без этой проверки тест тихо зеленеет на отсутствующей папке: rglob по
    # несуществующему пути отдаёт пустоту, а не исключение.
    assert (ROOT / package).is_dir(), f"нет пакета {package}"

    found = []
    for path in sorted((ROOT / package).rglob("*.py")):
        for root in imported_roots(path):
            if root in banned:
                found.append(f"{path.relative_to(ROOT)} → {root}")
    return found


def test_backend_does_not_import_client():
    assert offenders("backend", {"client"}) == []


def test_client_does_not_import_storage_libraries():
    assert offenders("client", BANNED_IN_CLIENT) == []


def test_client_takes_only_the_boundary_types_from_the_backend():
    """Виджет, импортирующий сервис, лежит не там.

    Проверка по корневым именам это пропускает: `backend.model.link` —
    такой же `backend`, как и `backend.model`, а тянет за собой репозитории,
    сессию и синхронный поход к базе мимо рантайма. Разделение слоёв
    перестаёт быть договорённостью только тогда, когда его проверяют целиком."""
    assert (ROOT / "client").is_dir(), "нет пакета client"

    found = []
    for path in sorted((ROOT / "client").rglob("*.py")):
        for module in imported_modules(path):
            if not module.startswith("backend"):
                continue
            if any(module == name or module.startswith(name + ".")
                   for name in ALLOWED_FROM_BACKEND):
                continue
            found.append(f"{path.relative_to(ROOT)} → {module}")

    assert found == []
```

```python
from collections.abc import Mapping
from dataclasses import dataclass
from pathlib import Path

DEFAULT_DATA_DIR = "data"


@dataclass(frozen=True, slots=True)
class Config:
    data_dir: Path
    tmdb_key: str
    mdblist_key: str
    omdb_key: str
    poiskkino_key: str
    kinopoisk_unofficial_key: str
    log_level: str

    @property
    def library_db(self) -> Path:
        return self.data_dir / "library.db"

    @property
    def posters_db(self) -> Path:
        return self.data_dir / "posters.db"

    @property
    def log_file(self) -> Path:
        return self.data_dir / "app.log"


def load_config(env: Mapping[str, str], base_dir: Path) -> Config:
    override = env.get("DATA_DIR", "").strip()
    return Config(
        data_dir=Path(override) if override else base_dir / DEFAULT_DATA_DIR,
        tmdb_key=env.get("TMDB_API_KEY", ""),
        mdblist_key=env.get("MDBLIST_API_KEY", ""),
        omdb_key=env.get("OMDB_API_KEY", ""),
        poiskkino_key=env.get("POISKKINO_API_KEY", ""),
        kinopoisk_unofficial_key=env.get("KINOPOISK_UNOFFICIAL_API_KEY", ""),
        log_level=env.get("LOG_LEVEL", "").strip() or "INFO",
    )
```

```python
    IMDB = "imdb"

    TMDB = "tmdb"
    ROTTEN = "rotten"
    METACRITIC = "metacritic"
    STEAM = "steam"
```

```python
"""Разовый сбор игровых жанров: справочника в документации нет ни у Steam, ни у
GOG, но каждый отдаёт свой список отдельной ручкой.

Запуск: pipenv run python scripts/collect_game_genres.py
"""

import asyncio
import json

import httpx

HEADERS = {"User-Agent": "PersonalRating/0.1 (personal library app)"}

# Витрина отдаёт имена без числовых идентификаторов, а карточка игры — с ними.
# Поэтому список берётся из витрины, а номера добираются по играм-примерам.
STEAM_MENU = "https://store.steampowered.com/api/getgenrelist/"
STEAM_APP = "https://store.steampowered.com/api/appdetails"
GOG_CATALOG = "https://catalog.gog.com/v1/catalog"

EXAMPLES_PER_GENRE = 2


def example_appids(menu: list[dict]) -> list[int]:
    """Игры-примеры из витрины. У платформенных строк вместо списка лежит false."""
    appids = []
    for genre in menu:
        items = genre.get("items")
        if isinstance(items, list):
            appids.extend(item["id"] for item in items[:EXAMPLES_PER_GENRE])
    return list(dict.fromkeys(appids))


async def app_genres(client: httpx.AsyncClient, appid: int) -> list[dict]:
    response = await client.get(STEAM_APP, params={"appids": appid, "l": "russian"})
    if response.status_code != 200:
        # Иначе на троттлинге скрипт свалится с JSONDecodeError.
        print(f"steam {appid}: код {response.status_code}, пропущено")
        return []
    payload = response.json().get(str(appid), {})
    if not payload.get("success"):
        print(f"steam {appid}: success=false, пропущено")
        return []
    return payload["data"].get("genres", [])


async def steam_genres(client: httpx.AsyncClient) -> dict[str, str]:
    response = await client.get(STEAM_MENU, params={"l": "russian"})
    menu = response.json()["genres"]
    print(f"витрина Steam: {len(menu)} строк")

    found: dict[str, str] = {}
    for appid in example_appids(menu):
        for genre in await app_genres(client, appid):
            found[str(genre["id"])] = genre["description"]
        await asyncio.sleep(0.3)
    return found


async def gog_genres(client: httpx.AsyncClient) -> dict[str, str]:
    """Полный список отдаётся фасетом каталога — выборка по играм не нужна."""
    response = await client.get(GOG_CATALOG, params={"limit": "1", "locale": "ru-RU"})
    found: dict[str, str] = {}
    for genre in response.json()["filters"]["fullGenresList"]:
        found.setdefault(genre["slug"], genre["name"])
    return found


async def main() -> None:
    async with httpx.AsyncClient(headers=HEADERS, timeout=20) as client:
        steam = await steam_genres(client)
        gog = await gog_genres(client)

    print(json.dumps({"steam": steam, "gog": gog}, ensure_ascii=False, indent=2))
    print(f"\nSteam: {len(steam)} жанров, GOG: {len(gog)} жанров")


if __name__ == "__main__":
    asyncio.run(main())
```

```python
from pathlib import Path

from sqlalchemy import event
from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)


def fold(value: str | None) -> str | None:
    """Регистр для сравнения названий. Встроенный в SQLite lower() умеет
    только ASCII и «Ведьмак» оставляет как есть — а Python приводит его
    правильно. Поэтому сравниваем обе стороны одними правилами."""
    return value.lower() if value is not None else None


def create_engine(path: Path) -> AsyncEngine:
    path.parent.mkdir(parents=True, exist_ok=True)
    engine = create_async_engine(f"sqlite+aiosqlite:///{path}")

    @event.listens_for(engine.sync_engine, "connect")
    def prepare_connection(dbapi_connection, _record):
        # SQLite выключает внешние ключи на каждом подключении заново.
        cursor = dbapi_connection.cursor()
        cursor.execute("PRAGMA foreign_keys=ON")
        cursor.close()
        dbapi_connection.create_function("py_lower", 1, fold, deterministic=True)

    return engine


def session_factory(engine: AsyncEngine) -> async_sessionmaker[AsyncSession]:
    return async_sessionmaker(engine, expire_on_commit=False)
```

```python
import os
from logging.config import fileConfig
from pathlib import Path

from alembic import context
from sqlalchemy import create_engine, event, pool

from backend.config import load_config
from backend.storage.session import prepare_connection
from backend.storage.tables import metadata

config = context.config
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = metadata


def database_path() -> Path:
    # `-x db=<путь>` нужен тестам: они гоняют миграции по временному файлу.
    override = context.get_x_argument(as_dictionary=True).get("db")
    if override:
        return Path(override)
    settings = load_config(os.environ, base_dir=Path(__file__).resolve().parents[1])
    return settings.library_db


def database_url() -> str:
    # Alembic ходит синхронным драйвером, приложение — асинхронным: разница
    # только в префиксе. Папку под базу заводим здесь — родная create_engine
    # её не создаёт и падает с «unable to open database file».
    path = database_path()
    path.parent.mkdir(parents=True, exist_ok=True)
    return f"sqlite:///{path}"


def migrate() -> None:
    engine = create_engine(database_url(), poolclass=pool.NullPool)

    # Соединение готовится слушателем, а не запросом на нём: любой execute
    # открывает транзакцию, Alembic считает её внешней и перестаёт коммитить
    # сам — миграции проходят, а данные откатываются, и alembic_version
    # остаётся пустым.
```

```python
def table_shape(inspector, name: str) -> dict:
    """Всё о таблице, что видно снаружи: не только имена колонок."""
    return {
        "columns": [
            (column["name"], str(column["type"]), column["nullable"])
            for column in inspector.get_columns(name)
        ],
        "primary_key": inspector.get_pk_constraint(name)["constrained_columns"],
        "unique": sorted(
            tuple(one["column_names"]) for one in inspector.get_unique_constraints(name)
        ),
        "indexes": sorted(
            (one["name"], tuple(one["column_names"]), one["unique"])
            for one in inspector.get_indexes(name)
        ),
        "foreign_keys": sorted(
            (
                tuple(one["constrained_columns"]),
                one["referred_table"],
                (one.get("options") or {}).get("ondelete"),
            )
            for one in inspector.get_foreign_keys(name)
        ),
    }


def shape(db_path: Path) -> dict:
    """Снимок всей базы: таблица за таблицей, кроме служебной alembic_version."""
    engine = create_engine(f"sqlite:///{db_path}")
    with engine.connect() as connection:
        inspector = inspect(connection)
        found = {
            name: table_shape(inspector, name)
            for name in sorted(inspector.get_table_names())
            if name != STAMP
        }
    engine.dispose()
    return found

    engine = create_engine(f"sqlite:///{db_path}")
    with engine.connect() as connection:
        found = connection.execute(
            text(f"SELECT version_num FROM {STAMP}")
        ).scalars().all()
    engine.dispose()
    return found
```

```python
from sqlalchemy import Row, func, select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker

from backend.model import Item, Kind, Source
from backend.storage.session import fold
from backend.storage.tables import item as item_table


class ItemExists(Exception):
    """Пара (source, source_id) уже занята другой карточкой."""


_FIELDS = (
    "kind", "source", "source_id", "title", "original_title", "year",
    "released_at", "synopsis", "poster_url", "runtime", "seasons_count",
    "episodes_count", "synced_at",
)


def _values(item: Item) -> dict:
    values = {name: getattr(item, name) for name in _FIELDS}
    values["kind"] = str(item.kind)
    values["source"] = str(item.source)
    return values


def _to_item(row: Row) -> Item:
    return Item(
        id=row.id,
        kind=Kind(row.kind),
        source=Source(row.source),
        source_id=row.source_id,
        title=row.title,
        synced_at=row.synced_at,
        original_title=row.original_title or "",
        year=row.year,
        released_at=row.released_at,
        synopsis=row.synopsis or "",
        poster_url=row.poster_url or "",
        runtime=row.runtime,
        seasons_count=row.seasons_count,
        episodes_count=row.episodes_count,
    )


class LibraryRepository:
    def __init__(self, sessions: async_sessionmaker[AsyncSession]) -> None:
        self._sessions = sessions

    async def create(self, item: Item) -> int:
        async with self._sessions() as session:
            try:
                result = await session.execute(
                    item_table.insert()
                    .values(**_values(item))
                    .returning(item_table.c.id)
                )
                await session.commit()
            except IntegrityError as error:
                await session.rollback()
                raise ItemExists(
                    f"{item.source}:{item.source_id} уже в библиотеке"
                ) from error
            return result.scalar_one()

    async def get(self, item_id: int) -> Item | None:
        async with self._sessions() as session:
            result = await session.execute(
                select(item_table).where(item_table.c.id == item_id)
            )
            row = result.one_or_none()
            return _to_item(row) if row else None

    async def find_by_source(self, source: Source, source_id: str) -> Item | None:
        async with self._sessions() as session:
            result = await session.execute(
                select(item_table).where(
                    item_table.c.source == str(source),
                    item_table.c.source_id == source_id,
                )
            )
            row = result.one_or_none()
            return _to_item(row) if row else None

    async def list_by_kind(self, kind: Kind, limit: int, offset: int) -> list[Item]:
        """Порядок по названию — решение этого плана, а не спеки: у разделов
        свои порядки, и они лягут в запросы сервисного слоя. Здесь нужен
        любой устойчивый, чтобы порции не перемешивались между запросами."""
        async with self._sessions() as session:
            result = await session.execute(
                select(item_table)
                .where(item_table.c.kind == str(kind))
                .order_by(item_table.c.title)
                .limit(limit)
                .offset(offset)
            )
            return [_to_item(row) for row in result.all()]

    async def find_by_title(self, kind: Kind, title: str) -> list[Item]:
        """Для предупреждения о совпадении: сравнение по обоим названиям и
        только внутри своего типа контента.

        Обе стороны приводятся одной и той же функцией: встроенный в SQLite
        lower() умеет только ASCII, и с ним русские названия не совпали бы
        никогда."""
        needle = fold(title.strip())
        async with self._sessions() as session:
            result = await session.execute(
                select(item_table).where(
                    item_table.c.kind == str(kind),
                    (func.py_lower(item_table.c.title) == needle)
                    | (func.py_lower(item_table.c.original_title) == needle),
                )
            )
            return [_to_item(row) for row in result.all()]
```

```python
from datetime import datetime

from sqlalchemy import delete, select
from sqlalchemy.dialects.sqlite import insert
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker

from backend.model import Rating, average_score
from backend.storage.tables import rating as rating_table


class RatingsRepository:
    def __init__(self, sessions: async_sessionmaker[AsyncSession]) -> None:
        self._sessions = sessions

    async def _write(self, item_id: int, values: dict, now: datetime) -> None:
        statement = insert(rating_table).values(
            item_id=item_id, created_at=now, updated_at=now, **values
        )
        statement = statement.on_conflict_do_update(
            index_elements=[rating_table.c.item_id],
            # created_at не трогаем: он про первую оценку, а не про последнюю.
            set_={**values, "updated_at": now},
        )
        async with self._sessions() as session:
            await session.execute(statement)
            await session.commit()

    async def set_score(self, item_id: int, score: float | None,
                        now: datetime) -> None:
        await self._write(item_id, {"score": score}, now)

    async def set_aspects(self, item_id: int, visual: int | None,
                          narrative: int | None, gameplay: int | None,
                          now: datetime) -> float | None:
        score = average_score(visual, narrative, gameplay)
        await self._write(
            item_id,
            {"score": score, "visual": visual, "narrative": narrative,
             "gameplay": gameplay},
            now,
        )
        return score

    async def set_review(self, item_id: int, review: str, now: datetime) -> None:
        await self._write(item_id, {"review": review}, now)

    async def get(self, item_id: int) -> Rating | None:
        async with self._sessions() as session:
            result = await session.execute(
                select(rating_table).where(rating_table.c.item_id == item_id)
            )
            row = result.one_or_none()
            if row is None:
                return None
            return Rating(
                item_id=row.item_id,
                score=row.score,
                visual=row.visual,
                narrative=row.narrative,
                gameplay=row.gameplay,
                review=row.review or "",
            )

    async def clear(self, item_id: int) -> None:
        """Снять оценку. Карточка остаётся — она уже загружена."""
        async with self._sessions() as session:
            await session.execute(
                delete(rating_table).where(rating_table.c.item_id == item_id)
            )
            await session.commit()
```

```python
from sqlalchemy.dialects.sqlite import insert
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker

from backend.model import Kind
from backend.storage.tables import item as item_table
from backend.storage.tables import item_tag as item_tag_table
from backend.storage.tables import tag as tag_table


class WrongTagKind(Exception):
    """Тег одного типа контента на карточке другого."""


class TagExists(Exception):
    """Тег с таким именем в этом наборе уже есть."""


class TagsRepository:
    def __init__(self, sessions: async_sessionmaker[AsyncSession]) -> None:
        self._sessions = sessions

    async def create(self, kind: Kind, name: str) -> int:
        async with self._sessions() as session:
            tail = await session.execute(
                select(func.coalesce(func.max(tag_table.c.position), -1))
                .where(tag_table.c.kind == str(kind))
            )
            try:
                result = await session.execute(
                    tag_table.insert()
                    .values(name=name, kind=str(kind), position=tail.scalar() + 1)
                    .returning(tag_table.c.id)
                )
                await session.commit()
            except IntegrityError as error:
                # Дубль имени внутри набора — своё исключение, а не чужое.
                # `IntegrityError` живёт в SQLAlchemy, и поймать его в клиенте
                # значит импортировать туда SQLAlchemy — то, что запрещено
                # границей слоёв и проверяется тестом задачи 1. Модалке тегов
                # надо показать «такой тег уже есть», и ловить она будет это.
                await session.rollback()
                raise TagExists(f"{kind}:{name} уже заведён") from error
            return result.scalar_one()

    async def list_for_kind(self, kind: Kind) -> list[tuple[int, str]]:
        async with self._sessions() as session:
            result = await session.execute(
                select(tag_table.c.id, tag_table.c.name)
                .where(tag_table.c.kind == str(kind))
                .order_by(tag_table.c.position)
            )
            return [(row.id, row.name) for row in result.all()]

    async def reorder(self, kind: Kind, tag_ids: list[int]) -> None:
        async with self._sessions() as session:
            for position, tag_id in enumerate(tag_ids):
                await session.execute(
                    update(tag_table)
                    .where(tag_table.c.id == tag_id, tag_table.c.kind == str(kind))
                    .values(position=position)
                )
            await session.commit()

    async def rename(self, tag_id: int, name: str) -> None:
        async with self._sessions() as session:
            await session.execute(
                update(tag_table).where(tag_table.c.id == tag_id).values(name=name)
            )
            await session.commit()

    async def delete(self, tag_id: int) -> None:
        async with self._sessions() as session:
            await session.execute(delete(tag_table).where(tag_table.c.id == tag_id))
            await session.commit()

    async def assign(self, item_id: int, tag_id: int) -> None:
        async with self._sessions() as session:
            kinds = await session.execute(
                select(item_table.c.kind, tag_table.c.kind.label("tag_kind"))
                .select_from(item_table)
                .join(tag_table, tag_table.c.id == tag_id, isouter=True)
                .where(item_table.c.id == item_id)
            )
            row = kinds.one_or_none()
            if row is None or row.tag_kind is None or row.kind != row.tag_kind:
                raise WrongTagKind(f"тег {tag_id} не для карточки {item_id}")

            await session.execute(
                insert(item_tag_table)
                .values(item_id=item_id, tag_id=tag_id)
                .on_conflict_do_nothing()
            )
            await session.commit()

    async def unassign(self, item_id: int, tag_id: int) -> None:
        async with self._sessions() as session:
            await session.execute(
                delete(item_tag_table).where(
                    item_tag_table.c.item_id == item_id,
                    item_tag_table.c.tag_id == tag_id,
                )
            )
            await session.commit()

    async def tags_of(self, item_id: int) -> list[tuple[int, str]]:
        async with self._sessions() as session:
            result = await session.execute(
                select(tag_table.c.id, tag_table.c.name)
                .join(item_tag_table, item_tag_table.c.tag_id == tag_table.c.id)
                .where(item_tag_table.c.item_id == item_id)
                .order_by(tag_table.c.position)
            )
            return [(row.id, row.name) for row in result.all()]
```

```python
import logging

from sqlalchemy import delete, select
from sqlalchemy.dialects.sqlite import insert
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker

from backend.model import Genre, GenreKind, Source
from backend.storage.tables import genre as genre_table
from backend.storage.tables import genre_map as genre_map_table
from backend.storage.tables import item_genre as item_genre_table

logger = logging.getLogger(__name__)


class GenresRepository:
    def __init__(self, sessions: async_sessionmaker[AsyncSession]) -> None:
        self._sessions = sessions

    async def resolve(self, source: Source, ext_ids: list[str]) -> list[int]:
        """Внешние идентификаторы в свои. Сборный разворачивается в два,
        незнакомый попадает в лог: имена в справочнике наши, а заводить жанр
        на лету нельзя."""
        if not ext_ids:
            return []

        async with self._sessions() as session:
            result = await session.execute(
                select(genre_map_table.c.ext_id, genre_map_table.c.genre_id)
                .where(
                    genre_map_table.c.source == str(source),
                    genre_map_table.c.ext_id.in_(ext_ids),
                )
            )
            rows = result.all()

        known: set[str] = set()
        resolved: list[int] = []
        for row in rows:
            known.add(row.ext_id)
            if row.genre_id not in resolved:
                resolved.append(row.genre_id)

        for ext_id in ext_ids:
            if ext_id not in known:
                logger.warning(
                    "неизвестный жанр: источник %s, идентификатор %s", source, ext_id
                )
        return resolved

    async def set_for_item(self, item_id: int, genre_ids: list[int]) -> None:
        async with self._sessions() as session:
            await session.execute(
                delete(item_genre_table).where(item_genre_table.c.item_id == item_id)
            )
            if genre_ids:
                await session.execute(
                    insert(item_genre_table).on_conflict_do_nothing(),
                    [{"item_id": item_id, "genre_id": genre_id}
                     for genre_id in genre_ids],
                )
            await session.commit()

    async def genres_of(self, item_id: int) -> list[Genre]:
        async with self._sessions() as session:
            result = await session.execute(
                select(genre_table)
                .join(item_genre_table,
                      item_genre_table.c.genre_id == genre_table.c.id)
                .where(item_genre_table.c.item_id == item_id)
                .order_by(genre_table.c.name)
            )
            return [
                Genre(id=row.id, name=row.name, kind=GenreKind(row.kind))
                for row in result.all()
            ]

    async def list_for_kind(self, kind: GenreKind) -> list[Genre]:
        async with self._sessions() as session:
            result = await session.execute(
                select(genre_table)
                .where(genre_table.c.kind == str(kind))
                .order_by(genre_table.c.name)
            )
            return [
                Genre(id=row.id, name=row.name, kind=GenreKind(row.kind))
                for row in result.all()
            ]
```

```python
from datetime import datetime
from pathlib import Path

from sqlalchemy import (
    BLOB,
    Column,
    DateTime,
    Integer,
    MetaData,
    Table,
    delete,
    select,
)
from sqlalchemy.dialects.sqlite import insert
from sqlalchemy.ext.asyncio import AsyncEngine

from backend.storage.session import create_engine

metadata = MetaData()

# Ключ — карточка, а не адрес: у залитой с диска картинки адреса нет, и
# скачанный постер от залитого отличаться не должен.
poster = Table(
    "poster", metadata,
    Column("item_id", Integer, primary_key=True),
    Column("data", BLOB, nullable=False),
    Column("fetched_at", DateTime, nullable=False),
)


class PosterStore:
    """Второй файл базы. Внешних ключей к основной здесь нет и быть не может:
    между двумя файлами SQLite их не бывает."""

    def __init__(self, path: Path) -> None:
        self._path = path
        self._engine: AsyncEngine | None = None

    async def open(self) -> None:
        self._engine = create_engine(self._path)
        async with self._engine.begin() as connection:
            await connection.run_sync(metadata.create_all)

    async def close(self) -> None:
        if self._engine is not None:
            await self._engine.dispose()
            self._engine = None

    async def put(self, item_id: int, data: bytes, now: datetime) -> None:
        statement = insert(poster).values(
            item_id=item_id, data=data, fetched_at=now
        )
        statement = statement.on_conflict_do_update(
            index_elements=[poster.c.item_id],
            set_={"data": statement.excluded.data,
                  "fetched_at": statement.excluded.fetched_at},
        )
        async with self._engine.begin() as connection:
            await connection.execute(statement)

    async def get(self, item_id: int) -> bytes | None:
        async with self._engine.connect() as connection:
            result = await connection.execute(
                select(poster.c.data).where(poster.c.item_id == item_id)
            )
            return result.scalar_one_or_none()

    async def has(self, item_id: int) -> bool:
        async with self._engine.connect() as connection:
            result = await connection.execute(
                select(poster.c.item_id).where(poster.c.item_id == item_id)
            )
            return result.one_or_none() is not None

    async def delete(self, item_id: int) -> None:
        async with self._engine.begin() as connection:
            await connection.execute(
                delete(poster).where(poster.c.item_id == item_id)
            )
```

## Обрезка постера

Решено: **Pillow, в бэкенде.** Залитая с диска картинка обрезается по центру до
2:3 там же, где хранится, — главный поток не трогается вовсе. Зависимость
добавляется в тот план, где появляется код обрезки, то есть в план 3; здесь
`PosterStore` по-прежнему хранит только байты.

Скачанному постеру обрезка не нужна: и TMDB `w500`, и Steam
`library_600x900_2x`, и GOG `coverVertical` уже 2:3.
